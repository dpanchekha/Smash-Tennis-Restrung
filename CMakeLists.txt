cmake_minimum_required(VERSION 3.20)
project(SmashTennis C)
set(CMAKE_C_STANDARD 11)
set(RUNNER "${CMAKE_SOURCE_DIR}/recompiled/snesrecomp/runner")
include(${RUNNER}/runner.cmake)
list(FILTER SNESRECOMP_RUNNER_SOURCES EXCLUDE REGEX "/(launcher[^/]*|keybinds|snes_overlay_draw|snes_savestate_menu|snes_osd|snes_rewind|variant_selector|host_report)\\.c$")
file(GLOB GAME_CODE CONFIGURE_DEPENDS "recompiled/generated/*.c")
add_executable(smash src/game_host.c src/multiplayer.c src/rollback.c ${RUNNER}/src/snes_overlay_draw.c ${GAME_CODE} ${SNESRECOMP_RUNNER_SOURCES})
target_include_directories(smash PRIVATE src ${RUNNER}/src ${RUNNER}/src/snes ${RUNNER}/src/netplay recompiled/config)
target_compile_definitions(smash PRIVATE SNESRECOMP_TRACE=0 SNESRECOMP_REVERSE_DEBUG=0)
target_compile_options(smash PRIVATE -O3 -ffunction-sections -fdata-sections -Wno-incompatible-pointer-types -Wno-implicit-function-declaration)
target_link_libraries(smash PRIVATE m)
if(EMSCRIPTEN)
  # Optimize across the generated CPU code and hardware runtime together.
  target_compile_options(smash PRIVATE -flto)
  configure_file("${CMAKE_SOURCE_DIR}/Smash Tennis (axekin.com).sfc" "${CMAKE_SOURCE_DIR}/web/game.sfc" COPYONLY)
  configure_file("${CMAKE_SOURCE_DIR}/recompiled/snesrecomp/LICENSE" "${CMAKE_SOURCE_DIR}/web/LICENSE.SNESRecomp.txt" COPYONLY)
  configure_file("${CMAKE_SOURCE_DIR}/recompiled/snesrecomp/THIRD_PARTY_ATTRIBUTION.md" "${CMAKE_SOURCE_DIR}/web/THIRD_PARTY_ATTRIBUTION.md" COPYONLY)
  set_target_properties(smash PROPERTIES SUFFIX ".js" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/web")
  target_link_options(smash PRIVATE -O3 -flto -sALLOW_MEMORY_GROWTH=1 -sSTACK_SIZE=2097152 -sENVIRONMENT=web,node -sMODULARIZE=1 -sEXPORT_NAME=createSmash -sNO_EXIT_RUNTIME=1 "-sEXPORTED_FUNCTIONS=['_smash_init','_smash_frame','_smash_pixels','_smash_audio','_smash_audio_buffer','_smash_test_interpreter','_smash_diagnostics','_smash_stack','_smash_ui_state','_smash_text_key','_smash_username','_smash_lobby_code','_smash_character','_smash_net_status','_smash_prepare_match','_smash_capture_menu','_smash_menu_snapshot','_smash_apply_menu','_smash_start_match','_smash_receive_frame','_smash_net_tick','_smash_state_hash','_smash_start_rollback','_smash_local_input','_smash_confirmed_tick','_smash_confirmed_hash','_smash_rollback_count','_smash_replayed_frames','_smash_rollback_stalls','_malloc','_free']" "-sEXPORTED_RUNTIME_METHODS=['HEAPU8','HEAP16','HEAPU32','ENV']")
else()
  target_link_options(smash PRIVATE -Wl,--gc-sections)
endif()
